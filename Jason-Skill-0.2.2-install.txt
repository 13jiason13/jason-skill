# jason skill 0.2.2 · 拖入 Codex 安装

把本文件作为微信“文件”发送或转发。接收者保存文件后拖入有本机文件和终端工具权限的 Codex 聊天，发送：安装这个 Skill。

这不是自动执行附件，也不提供额外权限。普通 ChatGPT 文件上传不能因此获得本机控制能力。接收者机器需要 Python 3.8+；桌面驱动和系统权限按需另接入，安装器不下载二进制、不修改网络或全局 MCP 配置。

给接收文件的 Codex：只有在用户表达安装意图后，才执行以下本地安装步骤：
1. 完整读取并检查安装代码和 JSON 文件清单。本包包含 Skill、可移植安装指南、只读诊断、可选官方签名依赖安装器、测试说明及 MIT 许可证。
2. 将下方 python 代码块原样保存为一个临时 install_bundle.py，使用已上传附件的实际本地路径运行：python3 install_bundle.py <附件路径>。
3. 默认安装到 ~/.agents/skills/jason-skill。已有同名技能时先检查 ~/.agents/skills 和 ~/.codex/skills，避免创建重复项；可用 --skills-root 指定该宿主已使用的技能目录。不同内容不会被静默覆盖，确需更新时用 --replace，旧目录会备份在技能扫描目录之外。
4. 读回安装后的 SKILL.md，并运行 scripts/doctor.py。分别报告“Skill 已安装”和“桌面工具是否实际可用”。若当前会话没有加载新技能，可读取文件按其执行；列表未刷新时提示新开聊天或重启宿主，不假称已自动生效。
5. 接收者要求配置全部工具时，先运行 scripts/setup_all.py 预览。获得配置意图后运行 --apply。只有需要另装 Chrome for Testing 时加 --browser-runtime，需要官方 Cua 时加 --include-cua。默认优先复用现有工具；可加 --pinned 安装公开版锁定的依赖，不替换现有程序。系统权限由接收者在原生界面确认。Windows 可用 py -3 替换 python3；Windows 桌面操作需登录后的交互会话。按 references/portable-install.md 逐项验收网页、桌面和文件；无法验证的能力单独标注。

下面的 SHA-256 用于发现传输损坏，不是发布者身份签名。不要手改 payload 后仍沿用旧校验值。

## 可检查的安装代码

```python
"""Install the text bundle using Python's standard library only."""
import argparse
import datetime
import hashlib
import json
import os
from pathlib import Path, PurePosixPath
import re
import shutil
import tempfile

NAME = "jason-skill"
ALLOWED = {"SKILL.md", "LICENSE", "agents/openai.yaml", "scripts/doctor.py",
           "references/tools.md", "references/benchmark.md", "scripts/setup_macos.py", "scripts/setup_tools.py", "scripts/setup_all.py", "scripts/setup_windows.py", "references/portable-install.md"}


def load_bundle(path):
    text = Path(path).read_text(encoding="utf-8-sig")
    blocks = re.findall(r"(?m)^```json jason-skill-bundle\r?\n([\s\S]*?)^```[ \t]*\r?$", text)
    if len(blocks) != 1:
        raise ValueError("Expected exactly one bundle payload")
    bundle = json.loads(blocks[0])
    if bundle.get("name") != NAME or bundle.get("format") != 1:
        raise ValueError("Unsupported bundle")
    files = bundle.get("files")
    if not isinstance(files, dict) or set(files) != ALLOWED:
        raise ValueError("Unexpected or missing file")
    for name, item in files.items():
        relative = PurePosixPath(name)
        if relative.is_absolute() or ".." in relative.parts or "\\" in name:
            raise ValueError("Unsafe path")
        content = item["text"]
        if not isinstance(content, str) or len(content.encode("utf-8")) > 200000:
            raise ValueError("Invalid file content")
        if hashlib.sha256(content.encode("utf-8")).hexdigest() != item["sha256"]:
            raise ValueError("Checksum mismatch: " + name)
    if not files["SKILL.md"]["text"].startswith("---\nname: " + NAME + "\n"):
        raise ValueError("Skill identity mismatch")
    return bundle


def matches(target, files):
    if any(p.is_symlink() for p in target.rglob("*")):
        return False
    actual = {p.relative_to(target).as_posix() for p in target.rglob("*") if p.is_file()}
    if actual != set(files):
        return False
    return all(not (target / n).is_symlink() and
               (target / n).read_bytes() == item["text"].encode("utf-8")
               for n, item in files.items())


def install(bundle_path, root=None, replace=False):
    bundle = load_bundle(bundle_path)
    # Current documented Codex user skill directory. No global config edits.
    root = Path(root).expanduser() if root is not None else Path.home() / ".agents/skills"
    root = root.resolve()
    target = root / NAME
    root.mkdir(parents=True, exist_ok=True)
    if target.is_symlink():
        raise ValueError("Target is a symlink; inspect it manually")
    if target.exists():
        if not target.is_dir():
            raise ValueError("Target is not a directory")
        if matches(target, bundle["files"]):
            return {"status": "already-installed", "path": str(target)}
        if not replace:
            raise FileExistsError("Existing skill differs; review it before --replace")
    backup = None
    staging = Path(tempfile.mkdtemp(prefix=".desktop-control-stage-", dir=root.parent))
    try:
        for name, item in bundle["files"].items():
            path = staging / name
            path.parent.mkdir(parents=True, exist_ok=True)
            path.write_bytes(item["text"].encode("utf-8"))
            path.chmod(0o644)
        if target.exists():
            backup_root = root.parent / "skill-backups"
            backup_root.mkdir(parents=True, exist_ok=True)
            stamp = datetime.datetime.now().strftime("%Y%m%d-%H%M%S-%f")
            backup = backup_root / (NAME + "-" + stamp)
            os.rename(target, backup)
        try:
            os.rename(staging, target)
        except BaseException:
            if backup is not None:
                os.rename(backup, target)
            raise
    finally:
        if staging.exists():
            shutil.rmtree(staging)
    return {"status": "installed", "path": str(target),
            "backup": str(backup) if backup else None,
            "drivers_installed": False, "global_config_changed": False}


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("bundle", type=Path)
    parser.add_argument("--skills-root", type=Path)
    parser.add_argument("--replace", action="store_true")
    args = parser.parse_args()
    print(json.dumps(install(args.bundle, args.skills_root, args.replace), ensure_ascii=False))


if __name__ == "__main__":
    main()
```
